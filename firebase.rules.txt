// Starter Firestore rules; review before deployment.
rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  function signedIn(){return request.auth != null;}
  function role(){return get(/databases/$(database)/documents/staff/$(request.auth.uid)).data.role;}
  function manager(){return signedIn() && role() in ['owner','admin'];}
  match /public/{doc=**}{allow read:if true;allow write:if manager();}
  match /staff/{uid}{allow read:if signedIn() && (request.auth.uid==uid || role()=='owner');allow write:if signedIn() && role()=='owner';}
  match /applications/{id}{allow create:if true;allow read,update,delete:if manager();}
  match /jerseyOrders/{id}{allow create:if true;allow read,update,delete:if manager();}
 }
}
