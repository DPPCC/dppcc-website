
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // =========================================
    // AUTHENTICATION AND STAFF PERMISSIONS
    // =========================================

    function signedIn() {
      return request.auth != null;
    }

    function owner() {
      return signedIn()
        && request.auth.token.email ==
          'dungsreepaschimparacricketclub@gmail.com';
    }

    function staffPath() {
      return /databases/$(database)/documents/staff/$(request.auth.token.email);
    }

    function staffExists() {
      return signedIn() && exists(staffPath());
    }

    function staffData() {
      return get(staffPath()).data;
    }

    function activeStaff() {
      return staffExists()
        && staffData().active == true
        && staffData().email == request.auth.token.email;
    }

    function staffRole(role) {
      return activeStaff() && staffData().role == role;
    }

    function admin() {
      return owner() || staffRole('Admin');
    }

    function manager() {
      return owner() || staffRole('Manager');
    }

    function captain() {
      return owner() || staffRole('Captain');
    }

    function playerManager() {
      return admin() || captain();
    }

    function matchManager() {
      return manager();
    }

    function newsManager() {
      return admin();
    }

    function scoreManager() {
      return captain();
    }

    function submissionManager() {
      return manager();
    }

    // =========================================
    // PLAYER DATA VALIDATION
    // =========================================

    function validPlayer(data) {
      return data.keys().hasAll([
          'name',
          'role',
          'jerseyNo',
          'entryNo',
          'battingStyle',
          'photoUrl',
          'createdAt',
          'createdBy',
          'updatedAt',
          'updatedBy'
        ])
        && data.keys().hasOnly([
          'name',
          'role',
          'jerseyNo',
          'entryNo',
          'battingStyle',
          'photoUrl',
          'createdAt',
          'createdBy',
          'updatedAt',
          'updatedBy'
        ])
        && data.name is string
        && data.name.size() > 0
        && data.name.size() <= 80
        && data.role in [
          'Batsman',
          'Bowler',
          'All-Rounder',
          'Wicket Keeper'
        ]
        && data.jerseyNo is int
        && data.jerseyNo >= 0
        && data.jerseyNo <= 999
        && data.entryNo is string
        && data.entryNo.size() <= 30
        && data.battingStyle in [
          'Right-hand',
          'Left-hand'
        ]
        && data.photoUrl is string
        && data.createdAt is timestamp
        && data.createdBy is string
        && data.updatedAt is timestamp
        && data.updatedBy is string;
    }

    // =========================================
    // STAFF MANAGEMENT
    // =========================================

    match /staff/{email} {

      allow get: if owner()
        || (
          signedIn()
          && request.auth.token.email == email
        );

      allow list: if owner();

      allow create, update: if owner()
        && request.resource.data.keys().hasAll([
          'email',
          'role',
          'active',
          'updatedAt',
          'createdAt'
        ])
        && request.resource.data.keys().hasOnly([
          'email',
          'role',
          'active',
          'updatedAt',
          'createdAt'
        ])
        && request.resource.data.email == email
        && request.resource.data.role in [
          'Admin',
          'Manager',
          'Captain'
        ]
        && request.resource.data.active is bool
        && request.resource.data.updatedAt == request.time
        && request.resource.data.createdAt is timestamp;

      allow delete: if owner();
    }

    // =========================================
    // PLAYER DIRECTORY
    // =========================================

    match /players/{playerId} {

      allow read: if true;

      allow create: if playerManager()
        && validPlayer(request.resource.data)
        && request.resource.data.createdAt == request.time
        && request.resource.data.createdBy ==
          request.auth.token.email
        && request.resource.data.updatedAt == request.time
        && request.resource.data.updatedBy ==
          request.auth.token.email;

      allow update: if playerManager()
        && validPlayer(request.resource.data)
        && request.resource.data.createdAt ==
          resource.data.createdAt
        && request.resource.data.createdBy ==
          resource.data.createdBy
        && request.resource.data.updatedAt == request.time
        && request.resource.data.updatedBy ==
          request.auth.token.email;

      allow delete: if owner();
    }

    // =========================================
    // MATCHES
    // =========================================

    match /matches/{matchId} {

      allow read: if true;

      allow create: if matchManager()
        && request.resource.data.keys().hasAll([
          'teamName',
          'opponent',
          'date',
          'venue',
          'status',
          'result',
          'overs',
          'matchType',
          'createdAt',
          'createdBy'
        ])
        && request.resource.data.keys().hasOnly([
          'teamName',
          'opponent',
          'date',
          'venue',
          'status',
          'result',
          'overs',
          'matchType',
          'createdAt',
          'createdBy'
        ])
        && request.resource.data.teamName is string
        && request.resource.data.opponent is string
        && request.resource.data.date is string
        && request.resource.data.venue is string
        && request.resource.data.status in [
          'Upcoming',
          'Live',
          'Completed'
        ]
        && request.resource.data.result is string
        && request.resource.data.overs is number
        && request.resource.data.matchType is string
        && request.resource.data.createdAt == request.time
        && request.resource.data.createdBy ==
          request.auth.token.email;

      // Managers can update normal match information.
      allow update: if (
        matchManager()
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            'teamName',
            'opponent',
            'date',
            'venue',
            'status',
            'result',
            'overs',
            'matchType',
            'updatedAt',
            'updatedBy'
          ])
        && request.resource.data.updatedAt == request.time
        && request.resource.data.updatedBy ==
          request.auth.token.email
      ) || (
        // Captains can only set an unfinished match to Live
        // while submitting live scoring updates.
        scoreManager()
        && resource.data.status in [
          'Upcoming',
          'Live'
        ]
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            'status',
            'updatedAt',
            'updatedBy'
          ])
        && request.resource.data.status == 'Live'
        && request.resource.data.updatedAt == request.time
        && request.resource.data.updatedBy ==
          request.auth.token.email
      );

      allow delete: if owner();
    }

    // =========================================
    // LIVE SCORES
    // =========================================

    match /liveScores/{scoreId} {

      allow read: if true;

      allow create: if scoreManager()
        && request.resource.data.keys().hasAll([
          'matchId',
          'matchLabel',
          'innings',
          'over',
          'ball',
          'batter',
          'bowler',
          'runs',
          'extrasType',
          'extrasRuns',
          'wicket',
          'dismissedPlayer',
          'commentary',
          'legalBall',
          'createdAt',
          'createdBy'
        ])
        && request.resource.data.keys().hasOnly([
          'matchId',
          'matchLabel',
          'innings',
          'over',
          'ball',
          'batter',
          'bowler',
          'runs',
          'extrasType',
          'extrasRuns',
          'wicket',
          'dismissedPlayer',
          'commentary',
          'legalBall',
          'createdAt',
          'createdBy'
        ])
        && request.resource.data.matchId is string
        && request.resource.data.matchLabel is string
        && request.resource.data.innings in [1, 2]
        && request.resource.data.over is int
        && request.resource.data.over >= 0
        && request.resource.data.ball is int
        && request.resource.data.ball >= 1
        && request.resource.data.ball <= 6
        && request.resource.data.batter is string
        && request.resource.data.bowler is string
        && request.resource.data.runs is int
        && request.resource.data.runs >= 0
        && request.resource.data.runs <= 6
        && request.resource.data.extrasType in [
          'None',
          'Wide',
          'No Ball',
          'Bye',
          'Leg Bye'
        ]
        && request.resource.data.extrasRuns is int
        && request.resource.data.extrasRuns >= 0
        && request.resource.data.wicket is bool
        && request.resource.data.dismissedPlayer is string
        && request.resource.data.commentary is string
        && request.resource.data.legalBall is bool
        && request.resource.data.createdAt == request.time
        && request.resource.data.createdBy ==
          request.auth.token.email;

      allow update, delete: if owner();
    }

    // =========================================
    // NEWS
    // =========================================

    match /news/{newsId} {

      allow read: if true;

      allow create: if newsManager()
        && request.resource.data.keys().hasAll([
          'title',
          'body',
          'createdAt',
          'createdBy'
        ])
        && request.resource.data.keys().hasOnly([
          'title',
          'body',
          'createdAt',
          'createdBy'
        ])
        && request.resource.data.title is string
        && request.resource.data.body is string
        && request.resource.data.createdAt == request.time
        && request.resource.data.createdBy ==
          request.auth.token.email;

      allow update: if newsManager()
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            'title',
            'body'
          ])
        && request.resource.data.title is string
        && request.resource.data.body is string;

      allow delete: if newsManager();
    }

    // =========================================
    // PLAYER REGISTRATIONS
    // =========================================

    match /registrations/{applicationId} {

      allow create: if request.resource.data.keys().hasAll([
          'name',
          'phone',
          'age',
          'role',
          'address',
          'photoUrl',
          'status',
          'createdAt'
        ])
        && request.resource.data.keys().hasOnly([
          'name',
          'phone',
          'age',
          'role',
          'address',
          'photoUrl',
          'status',
          'createdAt'
        ])
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0
        && request.resource.data.name.size() <= 100
        && request.resource.data.phone is string
        && request.resource.data.age is int
        && request.resource.data.age >= 8
        && request.resource.data.age <= 70
        && request.resource.data.role in [
          'Batsman',
          'Bowler',
          'All-Rounder',
          'Wicket Keeper'
        ]
        && request.resource.data.address is string
        && request.resource.data.photoUrl is string
        && request.resource.data.status == 'Pending'
        && request.resource.data.createdAt == request.time;

      allow read: if submissionManager();

      allow update: if submissionManager()
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            'status',
            'reviewedAt',
            'reviewedBy'
          ])
        && request.resource.data.status in [
          'Approved',
          'Rejected'
        ]
        && request.resource.data.reviewedAt == request.time
        && request.resource.data.reviewedBy ==
          request.auth.token.email;

      allow delete: if owner();
    }

    // =========================================
    // JERSEY ORDERS
    // =========================================

    match /jerseyOrders/{orderId} {

      allow create: if request.resource.data.keys().hasAll([
          'name',
          'phone',
          'size',
          'number',
          'createdAt'
        ])
        && request.resource.data.keys().hasOnly([
          'name',
          'phone',
          'size',
          'number',
          'status',
          'createdAt'
        ])
        && request.resource.data.name is string
        && request.resource.data.phone is string
        && request.resource.data.size in [
          'S',
          'M',
          'L',
          'XL',
          'XXL'
        ]
        && request.resource.data.number is int
        && request.resource.data.number >= 0
        && request.resource.data.number <= 999
        && (
          !('status' in request.resource.data)
          || request.resource.data.status == 'Pending'
        )
        && request.resource.data.createdAt == request.time;

      allow read: if submissionManager();

      allow update: if submissionManager()
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            'status',
            'statusUpdatedAt',
            'statusUpdatedBy'
          ])
        && request.resource.data.status in [
          'Pending',
          'Processing',
          'Completed',
          'Cancelled'
        ]
        && request.resource.data.statusUpdatedAt == request.time
        && request.resource.data.statusUpdatedBy ==
          request.auth.token.email;

      allow delete: if owner();
    }

    // =========================================
    // PLAYER QR VERIFICATION
    // QR records must be written by trusted
    // Admin SDK code, never directly by clients.
    // =========================================

    match /playerQrTokens/{token} {

      allow get: if token.matches('^[a-f0-9]{64}$');

      allow list: if false;

      // Cloud Functions using Firebase Admin SDK
      // bypass Firestore client security rules.
      allow create, update, delete: if false;
    }

    // =========================================
    // ADMIN ACTIVITY LOGS
    // Client-side writes disabled.
    // Trusted Admin SDK writes only.
    // =========================================

    match /activityLogs/{logId} {

      allow read: if owner();

      allow create, update, delete: if false;
    }

    // =========================================
    // DEFAULT DENY
    // =========================================

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
